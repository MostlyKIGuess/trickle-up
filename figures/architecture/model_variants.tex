% One member: standardize, regressor, optional gate/body MLP, sign classifier, outputs, recombination.
% #1 name prefix, #2 x origin, #3 1 to draw the gate/body MLP and the loss mask.
\newcommand{\member}[3]{
  \begin{scope}[shift={(#2,0)}]
    \ifnum#3=1 \def\wide{5.4}\else \def\wide{3.6}\fi
    \pgfmathsetmacro{\cx}{\wide/2}
    \node[tok=\wide cm] (#1-in) at (\cx,0) {Features: $V$, $T$, $W$, process};
    \node[lin=\wide cm] (#1-std) at (\cx,0.9) {Standardize};
    \node[mlp] (#1-reg) at (0.8,2.2) {Regressor\\[-1pt]{\tiny MLP $4{\times}256$}};
    \ifnum#3=1
      \node[new] (#1-gb) at (2.7,2.2) {Gate/body\\[-1pt]{\tiny MLP $4{\times}256$}};
      \node[cls] (#1-cls) at (4.6,2.2) {Sign\\[-1pt]classifier};
    \else
      \node[cls] (#1-cls) at (2.8,2.2) {Sign\\[-1pt]classifier};
    \fi
    % Regressor outputs log10|I| of the four terminals.
    \foreach \t [count=\i] in {d,g,s,b} {
      \node[sq=mlpD, fill=mlpL] (#1-y\t) at ($(0.8,3.45)+(\i*0.38-0.95,0)$) {\t};
    }
    \ifnum#3=1
      \node[sq=newD, fill=newL] (#1-zg) at ($(2.7,3.45)+(-0.19,0)$) {g};
      \node[sq=newD, fill=newL] (#1-zb) at ($(2.7,3.45)+(0.19,0)$) {b};
      \node[sq=clsD, fill=clsL] (#1-sd) at ($(4.6,3.45)+(-0.19,0)$) {d};
      \node[sq=clsD, fill=clsL] (#1-ss) at ($(4.6,3.45)+(0.19,0)$) {s};
    \else
      \node[sq=clsD, fill=clsL] (#1-sd) at ($(2.8,3.45)+(-0.19,0)$) {d};
      \node[sq=clsD, fill=clsL] (#1-ss) at ($(2.8,3.45)+(0.19,0)$) {s};
    \fi
    \node[lin=\wide cm] (#1-out) at (\cx,4.55) {$I = \mathrm{sign}\cdot 10^{\,y}$};
    % Wires.
    \draw[arr] (#1-in) -- (#1-std);
    \draw[arr] (#1-std.north -| #1-reg) -- (#1-reg);
    \draw[arr] (#1-std.north -| #1-cls) -- (#1-cls);
    \coordinate (#1-yt) at ($(#1-yg.north)!0.5!(#1-ys.north)$);
    \coordinate (#1-yu) at ($(#1-yg.south)!0.5!(#1-ys.south)$);
    \coordinate (#1-st) at ($(#1-sd.north)!0.5!(#1-ss.north)$);
    \coordinate (#1-su) at ($(#1-sd.south)!0.5!(#1-ss.south)$);
    \draw[arr] (#1-reg) -- (#1-yu);
    \draw[arr] (#1-cls) -- (#1-su);
    \draw[arr] (#1-yt) -- (#1-yt |- #1-out.south);
    \draw[arr] (#1-st) -- (#1-st |- #1-out.south);
    \node[sub, left=1pt of #1-yd] {$y$};
    \ifnum#3=1
      \coordinate (#1-zt) at ($(#1-zg.north)!0.5!(#1-zb.north)$);
      \coordinate (#1-zu) at ($(#1-zg.south)!0.5!(#1-zb.south)$);
      \draw[arr] (#1-std.north -| #1-gb) -- (#1-gb);
      \draw[arr] (#1-gb) -- (#1-zu);
      \draw[arr] (#1-zt) -- (#1-zt |- #1-out.south);
      \node[font=\tiny\bfseries, text=newD, anchor=south east, inner sep=1pt] at (#1-gb.north east) {ON};
      % OFF cases leave I_g at or below 1e-26 A out of the regressor loss.
      \node[draw=emph, dashed, line width=0.6pt, rounded corners=1pt, inner sep=1.2pt, fit=(#1-yg)] (#1-mask) {};
      \node[font=\tiny, text=emph, anchor=south, inner sep=1pt] at (#1-mask.north) {mask};
    \fi
  \end{scope}
}
